Supabase PostgreSQL DDL Schema

Skema SQL ini lengkap dengan Row Level Security (RLS) untuk deployment backend Supabase.

CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

CREATE TYPE user_role AS ENUM ('administrator', 'kepala_unit', 'supervisor', 'teknisi', 'cleaning_service', 'vendor');

CREATE TABLE public.profiles (
    id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,
    full_name TEXT NOT NULL,
    role user_role DEFAULT 'teknisi',
    phone TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE public.offices (
    id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
    code VARCHAR(20) UNIQUE NOT NULL,
    name VARCHAR(150) NOT NULL,
    address TEXT NOT NULL,
    pic_name VARCHAR(100),
    phone VARCHAR(30),
    building_area_sqm NUMERIC(10, 2),
    floor_count INT DEFAULT 1,
    status VARCHAR(20) DEFAULT 'Active'
);

ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public profiles viewable by auth users" ON public.profiles FOR SELECT USING (auth.role() = 'authenticated');

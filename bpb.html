Supabase SQL Schema & RLS Arsip

-- SUPABASE POSTGRESQL ARCHIVE & DOCUMENTS RLS
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

CREATE TABLE public.archives (
    id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
    title VARCHAR(200) NOT NULL,
    category VARCHAR(100) NOT NULL, -- Kontrak, SLF, Garansi, Utilitas, BAST
    office_id UUID REFERENCES public.offices(id),
    file_url TEXT NOT NULL,
    uploaded_by UUID REFERENCES public.profiles(id),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- RLS UNTUK AKSES ARSIP DOKUMEN
ALTER TABLE public.archives ENABLE ROW LEVEL SECURITY;

CREATE POLICY "View Archives Authenticated" 
ON public.archives FOR SELECT USING (auth.role() = 'authenticated');

CREATE POLICY "Manage Archives Admin Supervisor" 
ON public.archives FOR ALL USING (
    EXISTS (
        SELECT 1 FROM public.profiles 
        WHERE id = auth.uid() AND role IN ('administrator', 'supervisor', 'kepala_unit')
    )
);

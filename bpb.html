Supabase SQL Schema & RLS

-- SUPABASE POSTGRESQL RLS BBMS
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

CREATE TYPE user_role AS ENUM (
  'administrator', 'kepala_unit', 'supervisor', 
  'teknisi', 'cleaning_service', 'vendor'
);

CREATE TABLE public.profiles (
    id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,
    full_name TEXT NOT NULL,
    role user_role DEFAULT 'teknisi'
);

CREATE TABLE public.offices (
    id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
    code VARCHAR(20) UNIQUE NOT NULL,
    name VARCHAR(150) NOT NULL
);

ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public profiles viewable by auth users" 
ON public.profiles FOR SELECT USING (auth.role() = 'authenticated');
